#include <stdio.h>

// Check if allocation is possible with maxPages
int isPossible(int arr[], int n, int m, int maxPages) {
    int students = 1;
    int currentSum = 0;

    for (int i = 0; i < n; i++) {
        if (currentSum + arr[i] <= maxPages) {
            currentSum += arr[i];
        } else {
            students++;
            currentSum = arr[i];

            if (students > m)
                return 0;
        }
    }
    return 1;
}

int main() {
    int n, m;
    scanf("%d %d", &n, &m);

    int arr[n];
    int sum = 0, maxVal = 0;

    for (int i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
        sum += arr[i];
        if (arr[i] > maxVal)
            maxVal = arr[i];
    }

    int low = maxVal;
    int high = sum;
    int ans = sum;

    while (low <= high) {
        int mid = (low + high) / 2;

        if (isPossible(arr, n, m, mid)) {
            ans = mid;       // valid
            high = mid - 1;  // try smaller
        } else {
            low = mid + 1;   // increase limit
        }
    }

    printf("%d\n", ans);
    return 0;
}
