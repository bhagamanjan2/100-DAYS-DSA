#include <stdio.h>
#include <stdlib.h>

// Insertion sort for each bucket
void insertionSort(float arr[], int n) {
    for (int i = 1; i < n; i++) {
        float key = arr[i];
        int j = i - 1;

        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            j--;
        }
        arr[j + 1] = key;
    }
}

int main() {
    int n;
    scanf("%d", &n);

    float arr[n];
    for (int i = 0; i < n; i++)
        scanf("%f", &arr[i]);

    // Create buckets
    float buckets[n][n];   // worst-case
    int size[n];

    for (int i = 0; i < n; i++)
        size[i] = 0;

    // Distribute elements into buckets
    for (int i = 0; i < n; i++) {
        int index = (int)(n * arr[i]);
        buckets[index][size[index]++] = arr[i];
    }

    // Sort each bucket
    for (int i = 0; i < n; i++) {
        insertionSort(buckets[i], size[i]);
    }

    // Concatenate buckets
    int k = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < size[i]; j++) {
            arr[k++] = buckets[i][j];
        }
    }

    // Print sorted array
    for (int i = 0; i < n; i++) {
        printf("%.2f ", arr[i]);
    }

    return 0;
}
