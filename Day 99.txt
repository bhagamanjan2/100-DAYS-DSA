#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int start, end;
} Interval;

// Comparator for sorting by start time
int cmp(const void *a, const void *b) {
    Interval *x = (Interval *)a;
    Interval *y = (Interval *)b;
    return x->start - y->start;
}

int main() {
    int n;
    scanf("%d", &n);

    Interval arr[n];

    for (int i = 0; i < n; i++) {
        scanf("%d %d", &arr[i].start, &arr[i].end);
    }

    // Step 1: Sort intervals
    qsort(arr, n, sizeof(Interval), cmp);

    // Result array
    Interval result[n];
    int k = 0;

    result[k++] = arr[0];

    for (int i = 1; i < n; i++) {
        Interval *last = &result[k - 1];

        if (arr[i].start <= last->end) {
            // Merge
            if (arr[i].end > last->end)
                last->end = arr[i].end;
        } else {
            result[k++] = arr[i];
        }
    }

    // Print merged intervals
    for (int i = 0; i < k; i++) {
        printf("%d %d\n", result[i].start, result[i].end);
    }

    return 0;
}
