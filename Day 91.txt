#include <stdio.h>

// Check if possible within given time (max load per painter)
int isPossible(int arr[], int n, int k, int maxTime) {
    int painters = 1;
    int currentSum = 0;

    for (int i = 0; i < n; i++) {
        if (currentSum + arr[i] <= maxTime) {
            currentSum += arr[i];
        } else {
            painters++;
            currentSum = arr[i];

            if (painters > k)
                return 0;
        }
    }
    return 1;
}

int main() {
    int n, k;
    scanf("%d %d", &n, &k);

    int arr[n];
    int sum = 0, maxVal = 0;

    for (int i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
        sum += arr[i];
        if (arr[i] > maxVal)
            maxVal = arr[i];
    }

    int low = maxVal;
    int high = sum;
    int ans = sum;

    while (low <= high) {
        int mid = (low + high) / 2;

        if (isPossible(arr, n, k, mid)) {
            ans = mid;       // feasible
            high = mid - 1;  // try smaller time
        } else {
            low = mid + 1;   // need more time
        }
    }

    printf("%d\n", ans);
    return 0;
}
